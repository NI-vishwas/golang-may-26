// Define a Blue Print
type Employee struct{
	ID int
	Name string
	Position string
	Active bool
}

func main(){

	emp1 := Employee{
		ID: 101,
		Name: "Vishwas",
		Position: "Consultant",
		Active: true,
	}

	fmt.Println("Employee Name: ", emp1.Name)

	emp1.Active = false

	// Anonymous Struct
	point := struct{
		X int
		Y int
	}{X: 10, Y: 20}

	fmt.Println("Point: ", point)

}
